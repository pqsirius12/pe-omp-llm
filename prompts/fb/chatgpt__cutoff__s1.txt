```c
/* common.h - shared helpers for every kernel program.
 * Nothing here is parallel: timing, deterministic input generation,
 * argument parsing and the one-line result report the harness parses. */
#ifndef COMMON_H
#define COMMON_H

#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

/* Wall-clock seconds from a monotonic clock (unaffected by clock changes). */
static inline double now_sec(void) {
    struct timespec t;
    clock_gettime(CLOCK_MONOTONIC, &t);
    return (double)t.tv_sec + (double)t.tv_nsec * 1e-9;
}

/* xorshift64: tiny deterministic generator, identical on Linux and macOS,
 * so every run and every LLM variant gets exactly the same input data. */
static uint64_t rng_state = 88172645463325252ULL;
static inline void rng_seed(uint64_t s) { rng_state = s ? s : 1; }
static inline uint64_t rng_next(void) {
    rng_state ^= rng_state << 13;
    rng_state ^= rng_state >> 7;
    rng_state ^= rng_state << 17;
    return rng_state;
}
/* Uniform double in [0, 1): top 53 bits of the generator. */
static inline double rng_unit(void) {
    return (double)(rng_next() >> 11) * (1.0 / 9007199254740992.0);
}

static inline long arg_long(int argc, char **argv, int i, long def) {
    return argc > i ? atol(argv[i]) : def;
}

static inline void *xmalloc(size_t n) {
    void *p = malloc(n);
    if (!p) { fprintf(stderr, "out of memory (%zu bytes)\n", n); exit(1); }
    return p;
}

/* One machine-readable line per run; the harness parses exactly this. */
static inline void report(const char *name, double secs, double checksum) {
    printf("kernel=%s time_s=%.6f checksum=%.12e\n", name, secs, checksum);
}

#endif

```

Program to parallelize:
```c
/* k15_cutoff: for each point, count neighbours within radius r, scanning
 * points sorted by x and stopping once x is out of range.
 * IRREGULAR: points are clustered, so points inside dense clusters scan
 * far more neighbours than isolated ones (like molecular dynamics). */
#include <math.h>
#include "common.h"

/* === KERNEL BEGIN === */
static void kernel(long n, double r, const double *x, const double *y, int *nb) {
    for (long i = 0; i < n; i++) {
        int c = 0;
        for (long j = i + 1; j < n && x[j] - x[i] <= r; j++) {
            double dx = x[j] - x[i], dy = y[j] - y[i];
            if (dx * dx + dy * dy <= r * r) c++;
        }
        for (long j = i - 1; j >= 0 && x[i] - x[j] <= r; j--) {
            double dx = x[i] - x[j], dy = y[i] - y[j];
            if (dx * dx + dy * dy <= r * r) c++;
        }
        nb[i] = c;
    }
}
/* === KERNEL END === */

static int cmp_x(const void *a, const void *b) {
    double d = ((const double *)a)[0] - ((const double *)b)[0];
    return (d > 0) - (d < 0);
}

int main(int argc, char **argv) {
    long n = arg_long(argc, argv, 1, 400000);
    double r = 0.002;
    double *xy = xmalloc(2 * n * sizeof *xy);
    rng_seed(15);
    /* 70% of points in 8 tight Gaussian-ish clusters, 30% spread uniformly */
    for (long i = 0; i < n; i++) {
        if (rng_unit() < 0.7) {
            int c = (int)(rng_next() % 8);
            double cx = 0.1 + 0.1 * c, cy = 0.5;
            double a = rng_unit() + rng_unit() + rng_unit() - 1.5;   /* ~bell */
            double b = rng_unit() + rng_unit() + rng_unit() - 1.5;
            xy[2 * i] = cx + 0.01 * a; xy[2 * i + 1] = cy + 0.01 * b;
        } else { xy[2 * i] = rng_unit(); xy[2 * i + 1] = rng_unit(); }
    }
    qsort(xy, n, 2 * sizeof *xy, cmp_x);   /* sort (x,y) pairs by x */
    double *x = xmalloc(n * sizeof *x), *y = xmalloc(n * sizeof *y);
    for (long i = 0; i < n; i++) { x[i] = xy[2 * i]; y[i] = xy[2 * i + 1]; }
    int *nb = xmalloc(n * sizeof *nb);

    double t0 = now_sec();
    kernel(n, r, x, y, nb);
    double t1 = now_sec();

    double sum = 0.0;
    for (long i = 0; i < n; i++) sum += nb[i];
    report("cutoff", t1 - t0, sum);
    free(xy); free(x); free(y); free(nb);
    return 0;
}
```
